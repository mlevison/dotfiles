/**
 * Per-file batch-upload orchestration (spec upr2a).
 *
 * Uploads a batch of files one at a time, isolating each in its own try/catch so
 * that one failure never aborts the rest of the batch. The actual upload work is
 * injected via `uploadOne`, keeping this helper free of storage/OCR dependencies
 * and therefore unit-testable.
 */

export interface UploadBatchHooks<T> {
	/** Called after each file uploads successfully, in file order. */
	onSuccess?: (result: T, file: File) => void;
	/** Called when a file fails, before its message is recorded (e.g. to log). */
	onError?: (file: File, error: unknown) => void;
}

export interface UploadBatchResult<T> {
	/** Results of the files that uploaded successfully, in file order. */
	successes: T[];
	/** Per-file failure messages, each shaped `"<name>: <reason>"`. */
	failures: string[];
	/** Combined multi-line summary to show the user, or null if nothing failed. */
	errorMessage: string | null;
}

/**
 * Process an upload batch, isolating each file so a single failure does not abort
 * the rest. Successes and failures are accumulated and returned; the caller owns
 * any DOM concerns (spinner flag, input reset).
 *
 * @param files - The files to upload, in the order they should be processed.
 * @param uploadOne - Injected dependency that uploads a single file.
 * @param hooks - Optional per-file callbacks (success/error side effects).
 */
export async function processUploadBatch<T>(
	files: Iterable<File>,
	uploadOne: (file: File) => Promise<T>,
	hooks: UploadBatchHooks<T> = {}
): Promise<UploadBatchResult<T>> {
	const fileList = Array.from(files);
	const successes: T[] = [];
	const failures: string[] = [];

	for (const file of fileList) {
		try {
			const result = await uploadOne(file);
			successes.push(result);
			hooks.onSuccess?.(result, file);
		} catch (error: unknown) {
			hooks.onError?.(file, error);
			const reason = error instanceof Error ? error.message : 'Failed to upload';
			failures.push(`${file.name}: ${reason}`);
		}
	}

	const errorMessage =
		failures.length > 0
			? `Failed ${failures.length} of ${fileList.length} receipt(s):\n${failures.join('\n')}`
			: null;

	return { successes, failures, errorMessage };
}
